import { Steps, TabItem, Tabs, Aside } from '@astrojs/starlight/components';

Vamos a medir una corrida sin salir de la compu. TorkeKit trae un **carro de calle simulado**: dos segundos quieto, después pasa el cuarto de milla, con el teléfono inclinado, ruido en los sensores y un GPS de 1 Hz, como uno de verdad.

<Steps>

1. **Creá la sesión y armala**

   `arm()` empieza a grabar. En una arrancada desde parado, el cronómetro arranca solo cuando el carro sale.

   <Tabs syncKey="platform">
   <TabItem label="Swift">

   ```swift
   import TorkeKit

   let session = RunSession()
   session.arm()
   ```

   </TabItem>
   <TabItem label="Kotlin">

   ```kotlin
   import today.torke.kit.session.RunSession
   import today.torke.kit.input.RecordedRunProvider

   val session = RunSession()
   session.arm()
   ```

   </TabItem>
   </Tabs>

2. **Dale la corrida simulada**

   `run` lee la fuente hasta el final y cierra la corrida. Con la corrida simulada tarda milisegundos.

   <Tabs syncKey="platform">
   <TabItem label="Swift">

   ```swift
   await session.run(with: RecordedRunProvider.demo())
   ```

   </TabItem>
   <TabItem label="Kotlin">

   ```kotlin
   session.run(RecordedRunProvider.demo())   // dentro de una corrutina
   ```

   </TabItem>
   </Tabs>

3. **Leé los tiempos**

   <Tabs syncKey="platform">
   <TabItem label="Swift">

   ```swift
   if let r = session.snapshot.result {
       print("60 ft:  \(r.sixtyFeet!) s")
       print("0–100:  \(r.zeroTo100!) s")
       print("1/8:    \(r.eighthMile!) s a \(r.eighthMileKmh!) km/h")
       print("1/4:    \(r.quarterMile!) s a \(r.quarterMileKmh!) km/h")
   }
   ```

   </TabItem>
   <TabItem label="Kotlin">

   ```kotlin
   session.snapshot.value.result?.let { r ->
       println("60 ft:  ${r.sixtyFeet} s")
       println("0–100:  ${r.zeroTo100} s")
       println("1/8:    ${r.eighthMile} s a ${r.eighthMileKmh} km/h")
       println("1/4:    ${r.quarterMile} s a ${r.quarterMileKmh} km/h")
   }
   ```

   </TabItem>
   </Tabs>

</Steps>

Tenés que ver esto, igual en iOS y en Android:

```text
60 ft:  2.100 s

1/8:    8.486 s a 145.8 km/h
```

<Aside type="note" title="Siempre la misma corrida">
`demo()` usa la semilla 42. La misma semilla da exactamente la misma corrida en las dos plataformas, así que sirve para pruebas automáticas: si el número cambia, cambió el motor. Probá `demo(seed: 7)` para otra corrida.
</Aside>

## Ahora con el teléfono

Cambiá la fuente por los sensores y escuchá los eventos. La sesión hace el resto: espera que el carro esté quieto, arma, detecta el arranque y termina sola.

<Tabs syncKey="platform">
<TabItem label="Swift">

```swift title="RunView.swift"
import SwiftUI
import TorkeKit
import TorkeKitSensors

struct RunView: View {
    @State private var session = RunSession()
    @State private var message = "Poné el carro en la línea"
    private let phone = PhoneSensorProvider()

    var body: some View {
        VStack(spacing: 12) {
            Text("\(Int(session.snapshot.speedKmh)) km/h")
                .font(.system(size: 64, weight: .black, design: .monospaced))
            Text(String(format: "%.2f s", session.snapshot.elapsed))
            Text(message).foregroundStyle(.secondary)
            Button("Armar") { session.arm() }
        }
        .task {
            await phone.requestPermission()
            session.start(with: phone)          // vista previa del GPS
            for await event in session.events() {
                switch event {
                case .armed: message = "Listo. Salí cuando quieras."
                case .disarmed: message = "Se movió: quedate quieto."
                case .finished(let r): message = "1/4: \(r.quarterMile ?? 0) s"
                case .failed(let f): message = "Sin corrida: \(f)"
                default: break
                }
            }
        }
        .onAppear { UIApplication.shared.isIdleTimerDisabled = true }
        .onDisappear { session.stop() }
    }
}
```

</TabItem>
<TabItem label="Kotlin">

```kotlin title="RunScreen.kt"
@Composable
fun RunScreen(phone: PhoneSensorProvider) {
    val scope = rememberCoroutineScope()
    val session = remember { RunSession() }
    val snapshot by session.snapshot.collectAsState()
    var message by remember { mutableStateOf("Poné el carro en la línea") }

    LaunchedEffect(Unit) {
        session.start(phone, scope)                // vista previa del GPS
        session.events.collect { event ->
            message = when (event) {
                RunEvent.Armed -> "Listo. Salí cuando quieras."
                RunEvent.Disarmed -> "Se movió: quedate quieto."
                is RunEvent.Finished -> "1/4: ${event.result.quarterMile} s"
                is RunEvent.Failed -> "Sin corrida: ${event.failure}"
                else -> message
            }
        }
    }
    DisposableEffect(Unit) { onDispose { session.stop() } }

    Column(horizontalAlignment = Alignment.CenterHorizontally) {
        Text("${snapshot.speedKmh.toInt()} km/h", fontSize = 64.sp, fontFamily = FontFamily.Monospace)
        Text("%.2f s".format(snapshot.elapsed))
        Text(message)
        Button(onClick = { session.arm() }) { Text("Armar") }
    }
}
```

</TabItem>
</Tabs>

<Aside type="caution" title="El soporte importa">
El teléfono tiene que ir **fijo** (un soporte rígido en el parabrisas o la consola), no en la mano ni suelto en el asiento. La orientación da igual: TorkeKit aprende hacia dónde es "adelante" en el arranque.
</Aside>
